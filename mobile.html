/* ========================================================= WALLORA GLOBAL THEME SYSTEM Same theme key as index.html ========================================================= */ const WALLORA_THEME_KEY = "wallora-theme"; function applyTheme(theme){ const button = document.getElementById("themeButton"); if(theme === "light"){ document.body.classList.add("light"); if(button){ button.innerText = "☀️"; } }else{ document.body.classList.remove("light"); if(button){ button.innerText = "🌙"; } } } /* Get current global Wallora theme */ function getWalloraTheme(){ const savedTheme = localStorage.getItem(WALLORA_THEME_KEY); return savedTheme === "light" ? "light" : "dark"; } /* Change theme */ function toggleTheme(){ const currentTheme = getWalloraTheme(); const newTheme = currentTheme === "light" ? "dark" : "light"; /* Save using SAME key as index.html */ localStorage.setItem( WALLORA_THEME_KEY, newTheme ); /* Apply immediately */ applyTheme(newTheme); } /* ========================================================= SYNC WHEN THEME CHANGES FROM ANOTHER PAGE/TAB ========================================================= */ window.addEventListener( "storage", function(event){ if( event.key === WALLORA_THEME_KEY && event.newValue ){ applyTheme( event.newValue ); } } ); /* ========================================================= RE-CHECK WHEN MOBILE PAGE BECOMES VISIBLE ========================================================= */ document.addEventListener( "visibilitychange", function(){ if(!document.hidden){ applyTheme( getWalloraTheme() ); } } ); /* ========================================================= RE-CHECK WHEN WINDOW GETS FOCUS ========================================================= */ window.addEventListener( "focus", function(){ applyTheme( getWalloraTheme() ); } ); /* ========================================================= INITIAL THEME ========================================================= */ applyTheme( getWalloraTheme() );
